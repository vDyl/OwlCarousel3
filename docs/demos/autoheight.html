Overview

Enable the plugin with autoHeight: true. It sets the stage height to the tallest currently visible item.

//default settings:
AutoHeight.Defaults = {
	autoHeight: false,
	autoHeightClass: 'owl-height'
};

Setup

$('.owl-carousel').owlCarousel({
	items:1,
	margin:10,
	autoHeight:true
});
