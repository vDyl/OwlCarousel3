Plugin API

Plugin API allows you to extend carousel object constructor and use internal functions and variables. Use callback events to communicate between host and plugin.

Plugin scaffolding:

/**
 * Plugin Name
 * @since 3.0.0
 */

;(function($, window, document, undefined) {
	'use strict';

	var PluginName = function(carousel) {
		this._core = carousel;
		this._handlers = {
			'initialized.owl.carousel': $.proxy(function(event) {
				if (event.namespace) {
					// Initialize the plugin here.
				}
			}, this)
		};

		this._core.options = $.extend({}, PluginName.Defaults, this._core.options);
		this._core.$element.on(this._handlers);
	};

	PluginName.Defaults = {
		optionName: 'value',
		optionName2: 'value'
	};

	PluginName.prototype.method = function() {};

	//destroy:
	PluginName.prototype.destroy = function() {
		for (var event in this._handlers) {
			this._core.$element.off(event, this._handlers[event]);
		}
	};

	$.fn.owlCarousel.Constructor.Plugins.PluginName = PluginName;

})(window.jQuery, window, document);
